import { PrismaClient } from '@prisma/client';

// Configura un cliente global en modo desarrollo para evitar reinstanciaciones
// debido al hot-reloading de herramientas como ts-node/nodemon.
const globalForPrisma = global as unknown as { prisma: PrismaClient };

export const prisma =
  globalForPrisma.prisma ||
  new PrismaClient({
    log: ['query', 'info', 'warn', 'error'], // Recomendado para depuración
  });

if (process.env.NODE_ENV !== 'production') {
  globalForPrisma.prisma = prisma;
}

// Exporta el cliente para que pueda ser importado y usado en tus servicios
export default prisma;
