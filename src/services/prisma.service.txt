// src/services/prisma.service.ts

import { PrismaClient } from '@/generated/prisma'; // Usa tu alias de ruta configurado

// 1. Declarar la variable para la instancia de Prisma.
//    Usamos 'let' para que pueda ser reasignada si es necesario, 
//    aunque generalmente se usa 'const' con un chequeo a nivel global.
//    La declaración 'globalThis' es una práctica común en Node.js 
//    para evitar que se creen nuevas instancias en el "Hot Reloading" 
//    durante el desarrollo con herramientas como Next.js o Vite.
let prisma: PrismaClient;

// 2. Comprobar si estamos en desarrollo (para usar globalThis) O si ya existe una instancia.
if (process.env.NODE_ENV === 'production') {
    // En producción, simplemente crea la instancia
    prisma = new PrismaClient();
} else {
    // En desarrollo: Usa el objeto global para preservar la instancia
    
    // Si la propiedad 'prisma' no existe en el objeto global, créala.
    // Esto previene que se creen múltiples instancias durante el Hot Reloading.
    if (!global.prisma) {
        global.prisma = new PrismaClient();
    }
    
    // Asigna la instancia global a nuestra variable local
    prisma = global.prisma;
}

// 3. Exportar la única instancia
export const prismaService = prisma;

// Nota: Para que el código anterior funcione, TypeScript necesita saber 
// que 'global' puede tener una propiedad llamada 'prisma'.
// (Ver Paso 4)
